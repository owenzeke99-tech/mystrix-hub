-- File lưu key ẩn 24 giờ
local SAVE_FILE_NAME = "MystrixHub_KeyCache.json"
local SESSION_DURATION = 86400 -- 24 giờ tính bằng giây

local function isKeyValid()
    if readfile and writefile and isfile and isfile(SAVE_FILE_NAME) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(SAVE_FILE_NAME))
        end)
        if success and data and data.Key == CORRECT_KEY then
            if os.time() - (data.Time or 0) < SESSION_DURATION then
                return true
            end
        end
    end
    return false
end

local function saveKeySession()
    if writefile then
        pcall(function()
            local data = {
                Key = CORRECT_KEY,
                Time = os.time()
            }
            writefile(SAVE_FILE_NAME, HttpService:JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ trong vòng 24h thì tự động load thẳng script chính ngầm

    print("Mystrix Hub Authenticated via Saved Session!")
    pcall(function()
        loadstring(game:HttpGet(MAIN_SCRIPT_URL))()
    end)
    return
end

-- Button Hover Effects
local function addHoverAnimation(btn, normalColor, hoverColor)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = hoverColor}):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = normalColor}):Play()
    end)
end

-- Smooth Shake Animation
local originalPosition = MainFrame.Position
local isShaking = false

local function smoothShake()
    if isShaking then return end
    isShaking = true
    
    local basePos = originalPosition
    local intensity = 10
    local speed = 0.035
    
    local shakes = {
        basePos + UDim2.new(0, intensity, 0, 0),
        basePos + UDim2.new(0, -intensity * 0.8, 0, 0),
        basePos + UDim2.new(0, intensity * 0.6, 0, 0),
        basePos + UDim2.new(0, -intensity * 0.4, 0, 0),
        basePos
    }
    
    for _, pos in ipairs(shakes) do
        TweenService:Create(MainFrame, TweenInfo.new(speed, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Position = pos}):Play()
        task.wait(speed)
    end
    
    MainFrame.Position = basePos
    isShaking = false
end
